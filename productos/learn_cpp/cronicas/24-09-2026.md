### [Exportación] ¿Diferencia entre devolver por \* &?  ==repaso==

```cpp

#include <iostream>
#include <string>

class Persona {
private:
    std::string* nombre; // El puntero dentro de la clase

public:
    // Constructor: asigna memoria dinámica para el nombre
    Persona(std::string nombreInicial) {
        nombre = new std::string(nombreInicial);
    }

    // Destructor: limpia la memoria para evitar fugas (Memory Leaks)
    ~Persona() {
        delete nombre;
    }

    // El método que retorna el puntero (*devolverNombre)
    std::string* devolverNombre() {
        return nombre; 
    }
};

int main() {
    // Creamos el objeto de la clase
    Persona usuario("Carlos");

    // =================================================================
    // CASO 1: Recibirlo en un PUNTERO (Apunta a la memoria original)
    // =================================================================
    std::string* ptrNombre = usuario.devolverNombre();
    std::cout << "Caso 1 (Puntero): " << *ptrNombre << std::endl;

    // Modificar a través del puntero altera el objeto original si este es público,
    // o te permite trabajar directamente con la misma dirección de memoria.
    *ptrNombre = "Carlos Modificado"; 


    // =================================================================
    // CASO 2: Recibirlo en una VARIABLE NORMAL (Copia por valor)
    // =================================================================
    // Usamos el asterisco (*) antes de la llamada para DESREFERENCIAR
    std::string copiaNombre = *usuario.devolverNombre();
    std::cout << "Caso 2 (Copia): " << copiaNombre << std::endl;

    // Modificar la copia NO afecta al objeto original
    copiaNombre = "Luis"; 


    // Verificación final del estado del objeto
    std::cout << "Nombre final en el objeto: " << *usuario.devolverNombre() << std::endl;

    return 0;
}


```
